import fitz, os, json, re, weasyprint
root = r'C:\Users\Surface Laptop 3\Downloads\SUT_Physics_Career_Guide_deliverables'
p12 = os.path.join(root, 'SUT_Physics_Career_Guide_v1.2.pdf')

# 1. portrait-only + page count
d = fitz.open(p12)
land = [i + 1 for i, p in enumerate(d) if p.rect.width > p.rect.height]
print('pages:', len(d), '| landscape:', land)
assert not land and 50 <= len(d) <= 60

# 2. immutables
fp = json.load(open('_immutable.json'))
for f, (sz, mt) in fp.items():
    q = os.path.join(root, f)
    assert os.path.getsize(q) == sz and int(os.path.getmtime(q)) == mt, f
print('v1/v1.1 untouched')

# 3. TOC folios match anchors
html = open('guide.html', encoding='utf-8').read()
doc = weasyprint.HTML(string=html, base_url='.').render()
pages = {}
for i, pg in enumerate(doc.pages):
    for name in pg.anchors:
        pages.setdefault(name, i)
main = pages['ch1']
FA = '۰۱۲۳۴۵۶۷۸۹'
fa = lambda n: ''.join(FA[int(x)] for x in str(n))
bad = []
for m in re.finditer(r'<a class="toc-l[12]" href="#([^"]+)">.*?<span class="n">([^<]*)</span></a>', html):
    a, shown = m.group(1), m.group(2)
    want = 'الف' if a == 'preface' else fa(pages[a] - main + 1)
    if shown != want:
        bad.append((a, shown, want))
print('TOC mismatches:', bad)
assert not bad

# 4. deliverables exist
assert os.path.exists(os.path.join(root, 'channel_post.md'))
print('ALL FINAL CHECKS PASS')
EOF
